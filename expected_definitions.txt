def NecessaryAgent.KnowsAll : (M : NecessaryAgent.Model) → M.Thing → Prop :=
fun M t =>
  M.Knows t NecessaryAgent.Content.nil ∧
    ∀ (r : M.Reality),
      M.InOmega r →
        M.Knows t (NecessaryAgent.Content.alt r) ∧
          ∀ (c : M.Consideration), M.Bears c r → M.Knows t (NecessaryAgent.Content.cons c r)
def NecessaryAgent.God : (M : NecessaryAgent.Model) → M.Thing → Prop :=
fun M N =>
  (∀ (w : M.W), M.Fundamental w N ∧ ∀ (t : M.Thing), M.Fundamental w t → t = N) ∧
    (∀ (w : M.W), (∃ x, M.ContingentItem x ∧ M.E w x) → M.BringsAbout w N) ∧
      M.Mind N ∧
        NecessaryAgent.KnowsAll M N ∧ M.Agent M.w₀ N ∧ M.P7Antecedent M.w₀ N ∧ M.ActsOnBest M.w₀ N ∧ M.AccordsValue M.w₀

def NecessaryAgent.Model.E : (M : NecessaryAgent.Model) → M.W → M.Item → Prop :=
fun M w x =>
  match x with
  | Sum.inl t => M.E_thing w t
  | Sum.inr s => M.E_state w s
def NecessaryAgent.Model.Anc : (M : NecessaryAgent.Model) → M.W → M.Item → M.Item → Prop :=
fun M w => Relation.TransGen (M.Causes w)
def NecessaryAgent.Model.Nec : (M : NecessaryAgent.Model) → M.Thing → Prop :=
fun M t => ∀ (w : M.W), M.E_thing w t
def NecessaryAgent.Model.NecState : (M : NecessaryAgent.Model) → M.State → Prop :=
fun M s => ∀ (w : M.W), M.E_thing w (M.bearer s) → M.E_state w s
def NecessaryAgent.Model.Concrete : (M : NecessaryAgent.Model) → M.Thing → Prop :=
fun M t => ∃ w y, M.Causes w (Sum.inl t) y
def NecessaryAgent.Model.ContingentItem : (M : NecessaryAgent.Model) → M.Item → Prop :=
fun M x =>
  match x with
  | Sum.inl t => ¬M.Nec t ∧ M.Concrete t
  | Sum.inr s => ¬M.NecState s ∧ M.Concrete (M.bearer s)
def NecessaryAgent.Model.FirstCont : (M : NecessaryAgent.Model) → M.W → M.Item → Prop :=
fun M w x => M.ContingentItem x ∧ M.E w x ∧ ∀ (y : M.Item), M.Anc w y x → ¬M.ContingentItem y
def NecessaryAgent.Model.F : (M : NecessaryAgent.Model) → M.W → NecessaryAgent.Pred M.Item :=
fun M w x => M.FirstCont w x
def NecessaryAgent.Model.Fundamental : (M : NecessaryAgent.Model) → M.W → M.Thing → Prop :=
fun M w t => M.Nec t ∧ M.Concrete t ∧ ∀ (y : M.Thing ⊕ M.State), ¬M.Causes w y (Sum.inl t)
def NecessaryAgent.Model.InGround : (M : NecessaryAgent.Model) → M.Item → Prop :=
fun M x =>
  match x with
  | Sum.inl t => M.Nec t ∧ M.Concrete t
  | Sum.inr s => M.NecState s ∧ M.Nec (M.bearer s) ∧ M.Concrete (M.bearer s)
def NecessaryAgent.Model.CauseOfF : (M : NecessaryAgent.Model) → M.W → M.Item → Prop :=
fun M w x => ∃ m, M.F w m ∧ M.Causes w x m
@[reducible] def NecessaryAgent.Model.Reality : NecessaryAgent.Model → Type :=
fun M => NecessaryAgent.Pred M.Item
def NecessaryAgent.Model.realOf : (M : NecessaryAgent.Model) → M.W → M.Reality :=
fun M w x => M.ContingentItem x ∧ M.E w x
def NecessaryAgent.Model.InOmega : (M : NecessaryAgent.Model) → M.Reality → Prop :=
fun M r => ∃ w, M.realOf w = r ∧ ∃ x, M.F w x
def NecessaryAgent.Model.InPhi : (M : NecessaryAgent.Model) → NecessaryAgent.Pred M.Item → Prop :=
fun M φ => ∃ w, M.F w = φ ∧ ∃ x, φ x
def NecessaryAgent.Model.emptyReality : (M : NecessaryAgent.Model) → M.Reality :=
fun M x => False
def NecessaryAgent.Model.Maximal : (M : NecessaryAgent.Model) → M.Reality → Prop :=
fun M r => (∀ (r' : M.Reality), M.InOmega r' → M.AtLeast r r') ∧ M.AtLeast r M.emptyReality
def NecessaryAgent.Model.AccordsValue : (M : NecessaryAgent.Model) → M.W → Prop :=
fun M w => M.Maximal (M.realOf w)
def NecessaryAgent.Model.Favored : (M : NecessaryAgent.Model) → M.W → Prop :=
fun M w => ∃ ch, M.Sens w M.Maximal ch
def NecessaryAgent.Model.SelectedBecause : (M : NecessaryAgent.Model) → M.W → M.State → Prop :=
fun M w s => M.Sens w M.Maximal (NecessaryAgent.Chan.state s)
def NecessaryAgent.Model.BringsAbout : (M : NecessaryAgent.Model) → M.W → M.Thing → Prop :=
fun M w t =>
  (∀ (y : M.Item), ¬M.Anc w y (Sum.inl t)) ∧ ∀ (x : M.Item), M.ContingentItem x → M.E w x → M.Anc w (Sum.inl t) x
def NecessaryAgent.Model.DeterministicWrt : (M : NecessaryAgent.Model) → {C : Type} → (M.W → C) → Prop :=
fun M {C} cfg => ∀ (w w' : M.W), cfg w = cfg w' → M.F w = M.F w'
def NecessaryAgent.Model.groundConfig : (M : NecessaryAgent.Model) →
  NecessaryAgent.Pred M.Thing × NecessaryAgent.Pred M.State :=
fun M => (fun t => M.Nec t ∧ M.Concrete t, fun s => M.InGround (Sum.inr s))
def NecessaryAgent.Model.Deterministic : NecessaryAgent.Model → Prop :=
fun M => M.DeterministicWrt fun x => M.groundConfig
def NecessaryAgent.Model.kindsOf : (M : NecessaryAgent.Model) → M.W → NecessaryAgent.Pred M.K :=
fun M w k => ∃ x, M.F w x ∧ M.kind x = k
def NecessaryAgent.Model.TypeDeterministic : NecessaryAgent.Model → Prop :=
fun M => ∀ (w w' : M.W), M.kindsOf w = M.kindsOf w'
def NecessaryAgent.Model.RepState : (M : NecessaryAgent.Model) → M.State → Prop :=
fun M s => ∃ c, M.Rep s c
def NecessaryAgent.Model.SelectingRep : (M : NecessaryAgent.Model) → M.W → M.State → Prop :=
fun M w s => (∃ m, M.F w m ∧ M.CausesVia w (M.bearer s) s m) ∧ M.RepState s ∧ M.SelectedBecause w s
def NecessaryAgent.Model.Agential : (M : NecessaryAgent.Model) → M.W → Prop :=
fun M w => M.Favored w ∧ ∃ s, M.SelectingRep w s
def NecessaryAgent.Model.Teleological : (M : NecessaryAgent.Model) → M.W → Prop :=
fun M w => M.Favored w ∧ ¬M.Agential w
def NecessaryAgent.Model.PropGoverned : (M : NecessaryAgent.Model) → M.W → Prop :=
fun M w => ¬M.Deterministic ∧ ¬M.Favored w ∧ M.HasProp
def NecessaryAgent.Model.Bare : (M : NecessaryAgent.Model) → M.W → Prop :=
fun M w => ¬M.Deterministic ∧ ¬M.Favored w ∧ ¬M.HasProp
def NecessaryAgent.Model.Derived : (M : NecessaryAgent.Model) → M.W → M.State → Prop :=
fun M w s => ∃ s', M.Source w s s'
def NecessaryAgent.Model.Historical : (M : NecessaryAgent.Model) → M.W → M.State → Prop :=
fun M w s => ∃ x, M.Anc w x (Sum.inr s) ∧ M.ContingentItem x
def NecessaryAgent.Model.RepAllOmega : (M : NecessaryAgent.Model) → M.State → Prop :=
fun M s =>
  M.Rep s NecessaryAgent.Content.nil ∧
    ∀ (r : M.Reality),
      M.InOmega r →
        M.Rep s (NecessaryAgent.Content.alt r) ∧
          ∀ (c : M.Consideration), M.Bears c r → M.Rep s (NecessaryAgent.Content.cons c r)
def NecessaryAgent.Model.Mind : (M : NecessaryAgent.Model) → M.Thing → Prop :=
fun M t => ∃ s, M.bearer s = t ∧ (∃ w, M.E_state w s) ∧ M.Mental s
def NecessaryAgent.Model.AccurateAll : (M : NecessaryAgent.Model) → M.State → Prop :=
fun M s =>
  ∀ (c : NecessaryAgent.Content (NecessaryAgent.Pred (M.Thing ⊕ M.State)) M.Consideration), M.Rep s c → M.Accurate s c
def NecessaryAgent.Model.Knows : (M : NecessaryAgent.Model) →
  M.Thing → NecessaryAgent.Content M.Reality M.Consideration → Prop :=
fun M t c => ∃ s, M.bearer s = t ∧ (∃ w, M.E_state w s) ∧ M.Rep s c ∧ M.Accurate s c
def NecessaryAgent.Model.Agent : (M : NecessaryAgent.Model) → M.W → M.Thing → Prop :=
fun M w t => ∃ s, M.bearer s = t ∧ M.SelectingRep w s

  (∃ s, M.bearer s = t ∧ M.SelectingRep w s ∧ M.RepAllOmega s ∧ M.AccurateAll s) ∧

def NecessaryAgent.SameProfile : (M : NecessaryAgent.Model) → M.Thing → M.Thing → Prop :=
fun M t t' =>
  (∀ (s : M.State),
      M.bearer s = t →
        M.NecState s →
          ∃ s',
            M.bearer s' = t' ∧
              M.NecState s' ∧
                (∀ (c : NecessaryAgent.Content (NecessaryAgent.Pred (M.Thing ⊕ M.State)) M.Consideration),
                    M.Rep s c ↔ M.Rep s' c) ∧
                  (M.Mental s ↔ M.Mental s') ∧
                    (∀ (c : NecessaryAgent.Content (NecessaryAgent.Pred (M.Thing ⊕ M.State)) M.Consideration),
                        M.Accurate s c ↔ M.Accurate s' c) ∧
                      ∀ (w : M.W) (P : NecessaryAgent.Pred (M.Thing ⊕ M.State) → Prop),
                        M.Sens w P (NecessaryAgent.Chan.state s) ↔ M.Sens w P (NecessaryAgent.Chan.state s')) ∧
    ∀ (s' : M.State),
      M.bearer s' = t' →
        M.NecState s' →
          ∃ s,
            M.bearer s = t ∧
              M.NecState s ∧
                (∀ (c : NecessaryAgent.Content (NecessaryAgent.Pred (M.Thing ⊕ M.State)) M.Consideration),
                    M.Rep s c ↔ M.Rep s' c) ∧
                  (M.Mental s ↔ M.Mental s') ∧
                    (∀ (c : NecessaryAgent.Content (NecessaryAgent.Pred (M.Thing ⊕ M.State)) M.Consideration),
                        M.Accurate s c ↔ M.Accurate s' c) ∧
                      ∀ (w : M.W) (P : NecessaryAgent.Pred (M.Thing ⊕ M.State) → Prop),
                        M.Sens w P (NecessaryAgent.Chan.state s) ↔ M.Sens w P (NecessaryAgent.Chan.state s')
